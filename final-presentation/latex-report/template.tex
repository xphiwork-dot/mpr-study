\section{Introduction}
\budget{~0.35 page (fill rest of Page 1 col 1)}

\begin{itemize}[leftmargin=*, nosep, topsep=2pt]
  \item \guide{Key Idea 1 --- Motivation} Contrast prior personalization benchmarks: \textbf{LaMP} (style/preference matching) and \textbf{PerLTQA / LongMemEval} (1--2 hop conversational recall with matched train/test distributions) vs.\ real-world multi-hop reasoning over $\sim$13,100 atomic private facts/user (e.g., Paper Fig.~1a: \emph{Alice $\rightarrow$ Bob $\rightarrow$ David $\rightarrow$ New York}).
  \item \guide{Key Idea 2 --- Explicit vs.\ Implicit Tension}
    \emph{Explicit (RAG):} keeps text in external indices (\texttt{BM25s}, \texttt{e5}), fails single-shot due to \emph{bridge-entity mismatch}.
    \emph{Implicit (LoRA SFT):} compiles facts into weights (\textsc{MaskSFT}, \textsc{AskSFT}), suffers capacity interference and reasoning erosion.
    \emph{Proposed Bridge:} \textsc{HybridMem} (\textsc{BlockSFT}) combines K-means clustered LoRA with RAG routing.
  \item \guide{Key Idea 3 --- Scope of Our Study} Explain our clean-room reproduction across 32,399 offline probes and six GPU sessions (\texttt{S1}--\texttt{S6}) on consumer hardware (Colab T4 4-bit 7B + laptop 0.5B), testing whether structural orderings and claims hold under paired bootstrap evaluation.
\end{itemize}

\section{Problem Setup and MPR Dataset}
\budget{~0.55 page (Page 1 col 2 + top of Page 2)}

\begin{itemize}[leftmargin=*, nosep, topsep=2pt]
  \item \guide{Key Idea 1 --- Formal Definition 1} Given private user statements $\mathcal{S} = \{s_1, \dots, s_n\}$ ($n \approx 13{,}100$) and multi-hop question $q$, predict $\hat{a} = f(q; \mathcal{S})$ scored by normalized Exact Match ($\text{EM}$). Highlight the \textbf{2 mandatory conditions}:
    (1) \emph{Personalized Necessity \& Sufficiency} ($\exists S \subseteq \mathcal{S}, S \neq \emptyset$ sufficient to deduce $a$, blocking world-knowledge shortcuts);
    (2) \emph{Multi-hop Irreducibility} ($\nexists s_i \in \mathcal{S}$ alone yields $a$).
  \item \guide{Key Idea 2 --- 5-Step KG Pipeline (Paper Fig.\ 2a)} Outline:
    (1) \emph{Meta-Graph} $\mathcal{G}$ (108 entity/attr spaces, 34 entity/attribute/value-offset relations) $\rightarrow$
    (2) \emph{12 User Graphs} $\tilde{G}$ ($\sim$5,902 nodes/user) $\rightarrow$
    (3) \emph{Statements} $\mathcal{S}$ ($\sim$13,097/user; 157,166 total) $\rightarrow$
    (4) \emph{Disambiguated Random Walks} ($k=2\text{--}10$ hops) $\rightarrow$
    (5) \emph{LLM QA Rewrite} $(T, M, q, a)$.
  \item \guide{Key Idea 3 --- Dataset Audit Note} Note the release count check: paper cites 108,000 (Table~2) and 10,800 (\S3.3.3), while the actual release holds \textbf{32,400 QAs} (12 users $\times$ 2,700 QAs; \textbf{32,399 valid} after 1 null in user~8 hop~4); paper eval uses a 2,700-Q subset.
\end{itemize}

\section{Methodology and Proposed \textsc{HybridMem}}
\budget{~0.95 page (Page 2 across col 1 \& col 2)}

\subsection{Memory Mechanisms and Reasoning Structures}
\begin{itemize}[leftmargin=*, nosep, topsep=2pt]
  \item \guide{Memory Taxonomy (Paper Fig.\ 2c)} Summarize:
    \emph{Explicit ($k=20$):} \textsc{SparseRAG} (\texttt{BM25s}), \textsc{DenseRAG} (\texttt{e5-base-v2}), \textsc{TreeRAG} (\texttt{MemTree}), \textsc{GraphRAG}, plus \textsc{Ignoramus} ($\hat{S}=\emptyset$) and \textsc{Oracle} ($\hat{S}=M$).
    \emph{Implicit (LoRA $r=8, \alpha=32$, epochs $1\text{--}10$):} \textsc{MaskSFT} (masked entity/relation prediction) and \textsc{AskSFT} (self-generated QA pairs).
  \item \guide{4 Reasoning Structures (Paper Fig.\ 2d)} Define flow \& call cost:
    \textbf{\textsc{NR}} (1 call, static top-$k$);
    \textbf{\textsc{SR}} ($L$ calls, default $L=5$, CoT thought $o_i$ + re-retrieval $\text{RAG}(q \Vert o_{1..i}, \mathcal{S})$);
    \textbf{\textsc{MR}} (12 calls at $L=5, B=2$, ToT branching $T=0.7$ + LLM selector);
    \textbf{\textsc{DR}} ($\sim$5.9 calls, Divide $\le 5$ sub-Qs without RAG $\rightarrow$ sequential Solve $\rightarrow$ Merge).
\end{itemize}

\subsection{Proposed Method: \textsc{HybridMem} (\textsc{BlockSFT})}
\begin{itemize}[leftmargin=*, nosep, topsep=2pt]
  \item \guide{Why Naive SFT+RAG (\texttt{MS+X}/\texttt{AS+X}) Fails} Explain three pitfalls: stuffing $\sim$13.1k facts into one LoRA causes knowledge interference, SFT erodes multi-step reasoning, and a single adapter is query-agnostic at inference.
  \item \guide{Offline Phase --- Clustering \& Per-Cluster LoRA} Embed user statements $\mathcal{S}$ via \texttt{e5-base-v2}, cluster via K-means into $C$ local context blocks ($C \in \{30, 50\}$), train an independent LoRA $\hat{\theta}_c$ per cluster alongside the text retrieval index.
  \item \guide{Online Phase --- Retrieval \& Majority-Vote Routing} Retrieve top-$k$ statements $\hat{S}$, select the best-matching adapter by majority vote $c^* = \arg\max_c \sum_{s \in \hat{S}} \mathbb{I}[\text{cluster}(s)=c]$, and run inference with $\hat{\theta}_{c^*}$ conditioned on $\hat{S}$.
\end{itemize}

\section{Experiments and Group Reproduction}
\budget{~1.15 pages (Page 2 end + Page 3)}

\subsection{Paper Findings Summary (8$\times$A800, 2,700 Qs)}
\begin{itemize}[leftmargin=*, nosep, topsep=2pt]
  \item \guide{Explicit Memory (Fig.\ 3)} \textsc{SR} \& \textsc{MR} outperform \textsc{NR} \& \textsc{DR} (+10--20\% range); \textsc{DenseRAG} leads short hops (semantic match), \textsc{SparseRAG} leads long hops (keyword match), \textsc{GraphRAG} lags due to entity noise.
  \item \guide{Ablations (Figs.\ 4--6)} Larger $k$ helps short hops while long hops peak at intermediate $k$; reasoning gains plateau after \textbf{3 steps}; \texttt{3B} drops sharply vs.\ \texttt{7B} on \textsc{NR}/\textsc{DR}.
  \item \guide{Implicit Collapse (Fig.\ 8)} Pure \textsc{MaskSFT}/\textsc{AskSFT} stays near 0\%; adding SFT to \textsc{Oracle} (\textsc{MSO}/\textsc{ASO}) hurts \textsc{SR}/\textsc{MR} reasoning.
  \item \guide{\textsc{HybridMem} Table~3 \& Sensitivity (Figs.\ 7, 9, 10)} Highlights \textsc{SR} \textsc{SparseRAG} long-hop gain ($0.200 \rightarrow 0.232$); cluster counts 30 vs.\ 50 are robust (50 slightly higher); optimal training at \textbf{epoch~7}; incurs clustering + adapter-swap latency.
\end{itemize}

\section{Independent Discussion and Analysis}
\budget{~0.60 page (Page 3 end / Page 4 col 1)}

\begin{itemize}[leftmargin=*, nosep, topsep=2pt]
  \item \guide{1. Why \textsc{DR} Collapses to \textsc{NR}} Analyze the 3 failure links from \texttt{S6} traces:
    (a) \emph{blind Divide} splits $q$ with zero retrieved facts $\rightarrow$ ungrounded sub-queries;
    (b) \emph{Solve$\rightarrow$Merge recall drop} (Merge gold recall 0.353/0.369 $<$ \textsc{NR} 0.466/0.497);
    (c) non-CoT ``concise'' sub-answers hurt arithmetic even under all-gold recall (0.242 vs.\ \textsc{MR} 0.375).
  \item \guide{2. Why \textsc{MR} Ties \textsc{SR} at $2.4\times$ Cost} Explain \texttt{S5} trace findings:
    (a) append-only candidate list leaves shallow depth 1--2 thoughts selected on \textbf{74.5\%} of Qs;
    (b) \textbf{32.3\%} (593/1,836) of branch pairs are exact duplicates, wasting 12 calls/Q ($2.05\times$ wall time) for $+0.010$/$-0.007$ $\Delta\text{EM}$.
  \item \guide{3. Prompt Numeric Leakage (53\% / Leak-Free Splits)} Explain how LLM rewriting copies gold numbers into $q$ (\textbf{53\%} overall, \textbf{82\%} at hop~10; 38\%/62\% for 3+ digits), inflating \texttt{s900} \textsc{Ignoramus} EM $4.6\times$ (0.065 vs.\ 0.014 leak-free) and masking true RAG gain (+6.8\,pp all $\rightarrow$ +10.7\,pp leak-free).
  \item \guide{4. Practical Agent Design Rules} Synthesize 3 rules: default to 3-step \textsc{SR}; never decompose before first retrieval; prefer hybrid lexical+dense explicit indices over per-cluster LoRA memorization.
\end{itemize}
